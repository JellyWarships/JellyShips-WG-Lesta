# QtanLive 配置 —— 本包强度：12倍
# amplitude : 拉伸幅度，一个轴 +a%%、另一个轴 -a%%（上限 0.95，再大会翻转）
# period    : 一次振铃周期（秒），越小抖得越快
# damping   : 阻尼，越大越"闷"；0 = 不衰减
# lean_deg  : 额外左右摇摆（度），0 = 关
# axes      : auto = 按包围盒判断（宽度 ↔ 高度）；也可写 0,1 或 1,0 强制
#
# 同一时间只装一个档位（1/5/8/12 倍），否则两份模组每帧互相覆盖 → 闪烁。

amplitude = 0.920
period = 0.55
damping = 2.4
lean_deg = 5.0
axes = auto
